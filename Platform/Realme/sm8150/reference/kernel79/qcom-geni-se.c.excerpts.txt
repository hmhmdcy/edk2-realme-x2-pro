473: /**
474:  * DOC: Overview
475:  *
476:  * GENI FIFO packing is highly configurable. TX/RX packing/unpacking consist
477:  * of up to 4 operations, each operation represented by 4 configuration vectors
478:  * of 10 bits programmed in GENI_TX_PACKING_CFG0 and GENI_TX_PACKING_CFG1 for
479:  * TX FIFO and in GENI_RX_PACKING_CFG0 and GENI_RX_PACKING_CFG1 for RX FIFO.
480:  * Refer to below examples for detailed bit-field description.
481:  *
482:  * Example 1: word_size = 7, packing_mode = 4 x 8, msb_to_lsb = 1
483:  *
484:  *        +-----------+-------+-------+-------+-------+
485:  *        |           | vec_0 | vec_1 | vec_2 | vec_3 |
486:  *        +-----------+-------+-------+-------+-------+
487:  *        | start     | 0x6   | 0xe   | 0x16  | 0x1e  |
488:  *        | direction | 1     | 1     | 1     | 1     |
489:  *        | length    | 6     | 6     | 6     | 6     |
490:  *        | stop      | 0     | 0     | 0     | 1     |
491:  *        +-----------+-------+-------+-------+-------+
492:  *
522: #define PACKING_VECTOR_SHIFT 10
523: /**
524:  * geni_se_config_packing() - Packing configuration of the serial engine
525:  * @se:		Pointer to the concerned serial engine
526:  * @bpw:	Bits of data per transfer word.
527:  * @pack_words:	Number of words per fifo element.
528:  * @msb_to_lsb:	Transfer from MSB to LSB or vice-versa.
529:  * @tx_cfg:	Flag to configure the TX Packing.
530:  * @rx_cfg:	Flag to configure the RX Packing.
531:  *
532:  * This function is used to configure the packing rules for the current
533:  * transfer.
534:  */
535: void geni_se_config_packing(struct geni_se *se, int bpw, int pack_words,
536: 			    bool msb_to_lsb, bool tx_cfg, bool rx_cfg)
537: {
538: 	u32 cfg0, cfg1, cfg[NUM_PACKING_VECTORS] = {0};
539: 	int len;
540: 	int temp_bpw = bpw;
541: 	int idx_start = msb_to_lsb ? bpw - 1 : 0;
542: 	int idx = idx_start;
543: 	int idx_delta = msb_to_lsb ? -BITS_PER_BYTE : BITS_PER_BYTE;
544: 	int ceil_bpw = ALIGN(bpw, BITS_PER_BYTE);
545: 	int iter = (ceil_bpw * pack_words) / BITS_PER_BYTE;
546: 	int i;
547: 
548: 	if (iter <= 0 || iter > NUM_PACKING_VECTORS)
549: 		return;
550: 
551: 	for (i = 0; i < iter; i++) {
552: 		len = min_t(int, temp_bpw, BITS_PER_BYTE) - 1;
553: 		cfg[i] = idx << PACKING_START_SHIFT;
554: 		cfg[i] |= msb_to_lsb << PACKING_DIR_SHIFT;
555: 		cfg[i] |= len << PACKING_LEN_SHIFT;
556: 
586: 		writel_relaxed(bpw / 16, se->base + SE_GENI_BYTE_GRAN);
587: }
